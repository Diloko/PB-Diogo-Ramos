SELECT DISTINCT * FROM "AwsDataCatalog"."data-lake-diogo"."refined" limit 10;

CREATE OR REPLACE VIEW refined_view AS
SELECT 
    titulooriginal,
    ano_lancamento,
    genero,
    original_title,
    revenue,
    profit,
    budget,
    (profit - budget) AS lucro,
    popularity
FROM "AwsDataCatalog"."data-lake-diogo"."refined";

SELECT DISTINCT * FROM "AwsDataCatalog"."data-lake-diogo"."refined_view" 
WHERE genero LIKE '%Comedy%'
    AND genero NOT LIKE '%Animation%'
    AND anoLancamento BETWEEN '2011' AND '2020'
    AND revenue <> 0
order by popularity desc
limit 10;

SELECT DISTINCT * FROM "AwsDataCatalog"."data-lake-diogo"."refined_view" 
WHERE genero LIKE '%Animation%'
    AND anoLancamento BETWEEN '2011' AND '2020'
    AND revenue <> 0
order by popularity desc
limit 10;
